    63→// ===== AI Vision Integration =====
    64→async function scanWithAI(imageBase64: string): Promise<any[]> {
    65→  const { createVlm } = await import('z-ai-web-dev-sdk')
    66→  const vlm = createVlm({ apiKey: process.env.NEXT_PUBLIC_ZAI_API_KEY || '' })
    67→  const result = await vlm.chat({
    68→    messages: [
    69→      {
    70→        role: 'user',
    71→        content: [
    72→          {
    73→            type: 'text',
    74→            text: 'Analyze this restaurant menu image. Extract all menu items. For each item, provide: name, description (brief), price (numeric only, handle Nigerian Naira ₦ symbol), and category. Return ONLY a JSON array of objects with fields: name, description, price, category. If you cannot determine a field, use empty string or 0. Example: [{"name":"Chicken Burger","description":"Crispy chicken patty","price":4500,"category":"Burgers"}]',
    75→          },
    76→          { type: 'image_url', image_url: { url: imageBase64 } },
    77→        ],
    78→      },
    79→    ],
    80→  })
    81→  const content = result.choices?.[0]?.message?.content || '[]'
    82→  const jsonMatch = content.match(/\[[\s\S]*\]/)
    83→  if (jsonMatch) {
    84→    return JSON.parse(jsonMatch[0])
    85→  }
    86→  return []
    87→}

    90→function fileToBase64(file: File): Promise<string> {
    91→  return new Promise((resolve, reject) => {
    92→    const reader = new FileReader()
    93→    reader.onload = () => resolve(reader.result as string)
    94→    reader.onerror = reject
    95→    reader.readAsDataURL(file)
    96→  })
    97→}
