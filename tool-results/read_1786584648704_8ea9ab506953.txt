    15→    64→async function scanWithAI(imageBase64: string): Promise<any[]> {
    16→    65→  const { createVlm } = await import('z-ai-web-dev-sdk')
    17→    66→  const vlm = createVlm({ apiKey: process.env.NEXT_PUBLIC_ZAI_API_KEY || '' })
    18→    67→  const result = await vlm.chat({
    19→    68→    messages: [
    20→    69→      {
    21→    70→        role: 'user',
    22→    71→        content: [
    23→    72→          {
    24→    73→            type: 'text',
    25→    74→            text: 'Analyze this restaurant menu image. Extract all menu items. For each item, provide: name, description (brief), price (numeric only, handle Nigerian Naira ₦ symbol), and category. Return ONLY a JSON array of objects with fields: name, description, price, category. If you cannot determine a field, use empty string or 0. Example: [{"name":"Chicken Burger","description":"Crispy chicken patty","price":4500,"category":"Burgers"}]',
    26→    75→          },
    27→    76→          { type: 'image_url', image_url: { url: imageBase64 } },
    28→    77→        ],
    29→    78→      },
    30→    79→    ],
    31→    80→  })
    32→    81→  const content = result.choices?.[0]?.message?.content || '[]'
    33→    82→  const jsonMatch = content.match(/\[[\s\S]*\]/)
    34→    83→  if (jsonMatch) {
    35→    84→    return JSON.parse(jsonMatch[0])
    36→    85→  }
    37→    86→  return []
    38→    87→}
